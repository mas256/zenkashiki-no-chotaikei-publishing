    \subsection{高難易度のもの}\label{節：高難易度のもの}
      ここでは複数の解法を組み合わせる問題,複数回の変換を要する問題,一見単純でありながら技巧的な変形を要する問題を扱う。

      \subsubsection{問題}\label{節：高難易度のもの：問題}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-advanced}{problem}{answer}]
          \item $\displaystyle a_1=b_1=1,$
            \[
              \begin{cases}
                a_{n+1}=2a_n+b_n+3^n,\\
                b_{n+1}=a_n+2b_n-3^n
              \end{cases}
              \qquad(n\ge1)
            \]
                とする。
                (1) \(a_n,b_n\) の一般項を求めよ。
                (2) \(\displaystyle\sum_{k=1}^{n}a_k\) を求めよ。
          \item $\displaystyle S_n=\sum_{k=1}^{n}a_k,\qquad
                a_1=1,\qquad
                a_{n+1}=4a_n-S_n$
                とする。\(a_n,S_n\) を求めよ。
          \item 正の数列 \(\{a_n\}\) を
            \[
              a_1=1,\qquad
              a_{n+1}=2^n\prod_{k=1}^{n}a_k
            \]
                で定める。\(a_n\) を求めよ。
          \item $\displaystyle a_1=\frac32,\qquad b_1=\frac12,$
            \[
              a_{n+1}=
              \frac32(a_n+b_n)+
              \frac12\frac{a_n-b_n}{1+a_n-b_n},
            \]
            \[
              b_{n+1}=
              \frac32(a_n+b_n)-
              \frac12\frac{a_n-b_n}{1+a_n-b_n}
            \]
                とする。\(a_n,b_n\) を求めよ。
          \item $\displaystyle a_1=2,\qquad
                a_{n+1}=1+\frac{a_n-1}{2+n(a_n-1)}$
                とする。
                (1) \(a_n\) を求めよ。
                (2) \(\displaystyle\lim_{n\to\infty}2^n(a_n-1)\) を求めよ。
          \item 正の数列 \(\{a_n\}\) が
            \[
              a_1=2,\qquad a_2=16,
            \]
            \[
              a_{n+2}a_n=2^{2n+1}a_{n+1}^2
            \]
                を満たす。\(a_n\) を求めよ。
          \item $\displaystyle a_1=3,\qquad
                a_{n+1}=
                \frac{(n+2)a_n-n}{na_n+2-n}$
                とする。\(a_n\) を求めよ。
          \item $\displaystyle a_1=\frac32,\qquad b_1=\frac12,$
            \[
              a_{n+1}=
              \frac52(a_n+b_n)+
              \frac{3(a_n-b_n)}
              {2\{3+(n+1)(a_n-b_n)\}},
            \]
            \[
              b_{n+1}=
              \frac52(a_n+b_n)-
              \frac{3(a_n-b_n)}
              {2\{3+(n+1)(a_n-b_n)\}}
            \]
                とする。\(a_n,b_n\) を求めよ。
          \item 正の数列 \(\{a_n\}\) を
            \[
              a_1=1,\qquad
              a_{n+1}
              =
              2^n
              \prod_{k=1}^{n}a_k^{\,n-k+1}
            \]
                で定める。\(a_n\) を求めよ。
          \item 正の数列 \(\{a_n\}\) が
            \[
              a_1=1,\qquad a_2=2,
            \]
            \[
              (na_n+a_{n+1})a_{n+2}
              =
              (n+1)a_{n+1}^2
            \]
                を満たす。\(a_n\) を求めよ。
          \item $\displaystyle a_1=3,\qquad
                a_{n+1}=\frac{a_n^2+1}{2a_n}$
                とする。\(a_n\) を求めよ。
          \item $\displaystyle a_1=2,\qquad
                a_{n+1}=2a_n^2-1$
                とする。\(a_n\) を求めよ。
          \item $\displaystyle a_1=2,\qquad
                a_{n+1}=\frac{a_n^2+3}{2a_n}$
                とする。\(a_n\) を求めよ。
          \item $\displaystyle a_1=3,\qquad
                a_{n+1}=a_n^2-2$
                とする。\(a_n\) を求めよ。
          \item $\displaystyle a_1=\frac13,\qquad
                a_{n+1}=\frac{2a_n}{1+a_n^2}$
                とする。\(a_n\) を求めよ。
        \end{enumerate}

      \subsubsection{方針}\label{節：高難易度のもの：方針}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-advanced}{hint}{problem}]
          \item 連立式の対称性から和と差に分離する。和は等比、差は階差型になる。最後に等比数列の和を使う。
          \item 部分和を直接解かず、\(S_n\) を消去して重解をもつ3項間漸化式へ落とす。
          \item 積を対数で総和へ変換し、隣接する式を引いて一次漸化式にする。
          \item 和と差に分離する。和は等比型、差は分数型となるのでさらに逆数化する。
          \item 不動点 \(1\) との差を取り、その逆数を新しい数列にする。すると一次非斉次漸化式になる。
          \item 対数化して2階差分を得た後、さらに階差を導入して平方和へ帰着する。
          \item まず2基準点 \(1,-1\) からの比を取り、それでも残る分数型を逆数化する。
          \item 和差で分離し、差側を逆数化して階差型にする。
          \item 積を対数で重み付き総和へ変換し、差分を2回取って3項間漸化式へ落とす。
          \item 3項非線形から階比を作り、逆数化し、不動点との差を取って階比型へ変換する。
          \item \(a_n\pm1\) を同時に変形し、その比を取ると平方型になる。
          \item \(a_n+\sqrt{a_n^2-1}\) を導入すると、漸化式が単純な平方へ変わる。
          \item \(\pm\sqrt3\) からの距離の比を取ると平方型になる。
          \item \(a_n=x+x^{-1}\) の形を作るため、\(\sqrt{a_n^2-4}\) を導入する。
          \item \(1\pm a_n\) を同時に因数分解し、その比を取ると平方型になる。
        \end{enumerate}

      \subsubsection{解答}\label{節：高難易度のもの：解答}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-advanced}{answer}{problem}]
          \item \[
              \begin{aligned}
                a_n &= \frac{5\cdot3^{n-1}-3}{2},\\
                b_n &= \frac{3-3^{n-1}}2
              \end{aligned}
            \]
            \[
              \sum_{k=1}^{n}a_k=
              \frac{5(3^n-1)-6n}{4}
            \]
          \item \[
              a_n=(n+1)2^{n-2},\qquad
              S_n=n2^{n-1}
            \]
          \item \[
              a_n=2^{\,2^{n-1}-1}
            \]
          \item \[
              \begin{aligned}
                a_n &= 3^{n-1}+\frac1{2n},\\
                b_n &= 3^{n-1}-\frac1{2n}
              \end{aligned}
            \]
          \item \[
              a_n=1+\frac1{3\cdot2^{n-1}-n-1}
            \]
            \[
              \lim_{n\to\infty}2^n(a_n-1)=\frac23
            \]
          \item \[
              a_n=
              2^{\frac{n(n-1)(2n-1)}6+2n-1}
            \]
          \item \[
              a_n=\frac{n^2-n+6}{n^2-n+2}
            \]
          \item \[
              \begin{aligned}
                a_n &= 5^{n-1}+\frac3{n^2+n+4},\\
                b_n &= 5^{n-1}-\frac3{n^2+n+4}
              \end{aligned}
            \]
          \item \[
              a_n=
              2^{
              \frac1{\sqrt5}
              \left[
              \left(\frac{3+\sqrt5}{2}\right)^{n-1}
              -
              \left(\frac{3-\sqrt5}{2}\right)^{n-1}
              \right]
              }
            \]
          \item \[
              a_n=
              \prod_{k=1}^{n-1}\frac{2k}{2k-1}
              =
              \frac{(2n-2)!!}{(2n-3)!!}
            \]
          \item \[
              a_n=
              \frac{2^{\,2^{n-1}}+1}
              {2^{\,2^{n-1}}-1}
            \]
          \item \[
              a_n=
              \frac12\left\{
              (2+\sqrt3)^{2^{n-1}}
              +
              (2-\sqrt3)^{2^{n-1}}
              \right\}
            \]
          \item \[
              a_n=
              \sqrt3\,
              \frac{1+(2-\sqrt3)^{2^n}}
              {1-(2-\sqrt3)^{2^n}}
            \]
          \item \[
              a_n=
              \left(\frac{3+\sqrt5}{2}\right)^{2^{n-1}}
              +
              \left(\frac{3-\sqrt5}{2}\right)^{2^{n-1}}
            \]
          \item \[
              a_n=
              \frac{2^{\,2^{n-1}}-1}
              {2^{\,2^{n-1}}+1}
            \]
        \end{enumerate}

      \subsubsection{解法}\label{節：高難易度のもの：解法}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-advanced}{solution}{problem}]
          \item \(s_n=a_n+b_n,\ d_n=a_n-b_n\) とおくと
            \[
              s_{n+1}=3s_n,\qquad d_{n+1}=d_n+2\cdot3^n.
            \]
                \(s_1=2,d_1=0\) より
            \[
              s_n=2\cdot3^{n-1},\qquad d_n=3^n-3.
            \]
                したがって
            \[
              a_n=\frac{s_n+d_n}{2}
              =\frac{5\cdot3^{n-1}-3}{2},
            \]
            \[
              b_n=\frac{s_n-d_n}{2}
              =\frac{3-3^{n-1}}2.
            \]
                さらに \(a_k\) を和に代入して
            \[
              \sum_{k=1}^{n}a_k
              =\frac12\left(5\frac{3^n-1}{2}-3n\right).
            \]
          \item $\displaystyle S_n=4a_n-a_{n+1}.$
                また
            \[
              a_{n+2}=4a_{n+1}-S_{n+1}
              =3a_{n+1}-S_n.
            \]
                したがって
            \[
              a_{n+2}=4a_{n+1}-4a_n.
            \]
                特性方程式は
            \[
              (r-2)^2=0.
            \]
                さらに \(a_2=3\) なので
            \[
              a_n=(A+B(n-1))2^{n-1}
            \]
                に初期条件を入れると \(A=1,B=\displaystyle \frac{1}{2}\). よって
            \[
              a_n=(n+1)2^{n-2}.
            \]
                最後に \(S_n=4a_n-a_{n+1}\) より
            \[
              S_n=n2^{n-1}.
            \]
          \item \(x_n=\log_2a_n\) とおくと
            \[
              x_{n+1}=n+\sum_{k=1}^{n}x_k.
            \]
                1つ前の式との差を取ると
            \[
              x_{n+1}=2x_n+1.
            \]
                \(x_1=0\) より
            \[
              x_n=2^{n-1}-1.
            \]
                したがって
            \[
              a_n=2^{2^{n-1}-1}.
            \]
          \item \(s_n=a_n+b_n,\ d_n=a_n-b_n\) とおくと
            \[
              s_{n+1}=3s_n,\qquad
              d_{n+1}=\frac{d_n}{1+d_n}.
            \]
                \(s_1=2\) より
            \[
              s_n=2\cdot3^{n-1}.
            \]
                また \(c_n=\displaystyle \frac{1}{d_n}\) とおくと
            \[
              c_{n+1}=c_n+1,\qquad c_1=1,
            \]
                したがって \(d_n=\displaystyle \frac{1}{n}\). ゆえに
            \[
              a_n=\frac{s_n+d_n}{2},\qquad
              b_n=\frac{s_n-d_n}{2}.
            \]
          \item $\displaystyle b_n=\displaystyle \frac1{a_n-1}$
                とおくと
            \[
              b_{n+1}=2b_n+n,\qquad b_1=1.
            \]
                特解を \(-n-1\) とすれば
            \[
              b_n=3\cdot2^{n-1}-n-1.
            \]
                よって
            \[
              a_n=1+\frac1{3\cdot2^{n-1}-n-1}.\]
                したがって
            \[
              2^n(a_n-1)
              =\frac{2^n}{3\cdot2^{n-1}-n-1}\to\frac23.
            \]
          \item \(x_n=\log_2a_n\) とおくと
            \[
              x_{n+2}-2x_{n+1}+x_n=2n+1.
            \]
                \(d_n=x_{n+1}-x_n\) とおけば
            \[
              d_{n+1}-d_n=2n+1.
            \]
                \(d_1=3\) より
            \[
              d_n=n^2+2.
            \]
                したがって
            \[
              x_n=1+\sum_{k=1}^{n-1}(k^2+2)
              =\frac{n(n-1)(2n-1)}6+2n-1.
            \]
          \item $\displaystyle b_n=\displaystyle \frac{a_n-1}{a_n+1}$
                とおくと
            \[
              b_{n+1}=\frac{b_n}{1+nb_n}.
            \]
                さらに \(c_n=\displaystyle \frac{1}{b_n}\) とおけば
            \[
              c_{n+1}=c_n+n.
            \]
                \(c_1=2\) より
            \[
              c_n=2+\frac{n(n-1)}2
              =\frac{n^2-n+4}{2}.
            \]
                したがって
            \[
              b_n=\frac2{n^2-n+4}.
            \]
                これを
            \[
              \frac{a_n-1}{a_n+1}=b_n
            \]
                へ戻せば答えを得る.
          \item \(s_n=a_n+b_n,\ d_n=a_n-b_n\) とおくと
            \[
              s_{n+1}=5s_n,\qquad
              d_{n+1}=\frac{3d_n}{3+(n+1)d_n}.
            \]
                よって
            \[
              s_n=2\cdot5^{n-1}.
            \]
                さらに \(c_n=\displaystyle \frac{1}{d_n}\) とおくと
            \[
              c_{n+1}=c_n+\frac{n+1}{3}.
            \]
                \(c_1=1\) なので
            \[
              c_n=\frac{n^2+n+4}{6},
              \qquad
              d_n=\frac6{n^2+n+4}.
            \]
                最後に
            \[
              a_n=(s_n+d_n)/2,\quad b_n=(s_n-d_n)/2.
            \]
          \item \(x_n=\log_2a_n\) とおくと
            \[
              x_{n+1}
              =n+\sum_{k=1}^{n}(n-k+1)x_k.
            \]
                差を取ると
            \[
              x_{n+1}-x_n=1+\sum_{k=1}^{n}x_k.
            \]
                さらにもう一度差を取ると
            \[
              x_{n+1}-3x_n+x_{n-1}=0.
            \]
                \(x_1=0,x_2=1\). 特性方程式
            \[
              r^2-3r+1=0
            \]
                の根を
            \[
              \alpha=\frac{3+\sqrt5}{2},\quad
              \beta=\frac{3-\sqrt5}{2}
            \]
                とすると
            \[
              x_n=\frac{\alpha^{n-1}-\beta^{n-1}}{\sqrt5}.
            \]
                よって \(a_n=2^{x_n}\).
          \item $\displaystyle r_n=\displaystyle \frac{a_{n+1}}{a_n}$
                とおくと
            \[
              r_{n+1}=\frac{(n+1)r_n}{n+r_n}.
            \]
                \(c_n=\displaystyle \frac{1}{r_n}\) とおけば
            \[
              c_{n+1}=\frac{nc_n+1}{n+1}.
            \]
                さらに \(d_n=c_n-1\) とおくと
            \[
              d_{n+1}=\frac{n}{n+1}d_n.
            \]
                \(r_1=2\) より \(d_1=-\displaystyle \frac{1}{2}\). したがって
            \[
              d_n=-\frac1{2n},
              \quad
              r_n=\frac{2n}{2n-1}.
            \]
                よって
            \[
              a_n=\prod_{k=1}^{n-1}\frac{2k}{2k-1}.
            \]
          \item \[
            \begin{aligned}
              a_{n+1}-1 &= \frac{(a_n-1)^2}{2a_n},\\
              a_{n+1}+1 &= \frac{(a_n+1)^2}{2a_n}.
            \end{aligned}
          \]
                したがって
            \[
              b_n=\frac{a_n-1}{a_n+1}
            \]
                とおくと
            \[
              b_{n+1}=b_n^2.
            \]
                \(b_1=\displaystyle \frac{1}{2}\) より
            \[
              b_n=2^{-2^{n-1}}.
            \]
                これを \(a_n\) について解く.
          \item $\displaystyle b_n=a_n+\sqrt{a_n^2-1}$
                とおく.
            \[
              a_{n+1}^2-1=4a_n^2(a_n^2-1)
            \]
                より
            \[
              b_{n+1}=b_n^2.
            \]
                \(b_1=2+\sqrt3\) だから
            \[
              b_n=(2+\sqrt3)^{2^{n-1}}.
            \]
                また
            \[
              a_n-\sqrt{a_n^2-1}=b_n^{-1},
            \]
                よって
            \[
              2a_n=b_n+b_n^{-1}.
            \]
          \item \[
            \begin{aligned}
              a_{n+1}-\sqrt3 &= \frac{(a_n-\sqrt3)^2}{2a_n},\\
              a_{n+1}+\sqrt3 &= \frac{(a_n+\sqrt3)^2}{2a_n}.
            \end{aligned}
          \]
                したがって
            \[
              b_n=\frac{a_n-\sqrt3}{a_n+\sqrt3}
            \]
                とおけば
            \[
              b_{n+1}=b_n^2.
            \]
                また
            \[
              b_1=\frac{2-\sqrt3}{2+\sqrt3}=(2-\sqrt3)^2.
            \]
                よって
            \[
              b_n=(2-\sqrt3)^{2^n}.
            \]
                最後に \(a_n\) について解く.
          \item \[
            b_n=\frac{a_n+\sqrt{a_n^2-4}}2
          \]
                とおく.
            \[
              a_{n+1}^2-4=a_n^2(a_n^2-4)
            \]
                より
            \[
              b_{n+1}=b_n^2.
            \]
                \(b_1=\displaystyle \frac{3+\sqrt5}{2}\) なので
            \[
              b_n=\left(\frac{3+\sqrt5}{2}\right)^{2^{n-1}}.
            \]
                またもう一方は \(b_n^{-1}\) だから
            \[
              a_n=b_n+b_n^{-1}.
            \]
          \item \[
            \begin{aligned}
              1-a_{n+1} &= \frac{(1-a_n)^2}{1+a_n^2},\\
              1+a_{n+1} &= \frac{(1+a_n)^2}{1+a_n^2}.
            \end{aligned}
          \]
                したがって
            \[
              b_n=\frac{1-a_n}{1+a_n}
            \]
                とおけば
            \[
              b_{n+1}=b_n^2.
            \]
                \(b_1=\displaystyle \frac{1}{2}\) より
            \[
              b_n=2^{-2^{n-1}}.
            \]
                これを \(a_n\) について解けばよい.
        \end{enumerate}

    \subsection{大学過去問ベースのもの}\label{節：大学過去問ベースのもの}
      以下の問題は大学入試問題の出題構造・発想を参考に,本書の演習として独立に再構成したものである。原典の問題文そのものではない。

      \subsubsection{問題}\label{節：大学過去問ベースのもの：問題}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-entrance}{problem}{answer}]
          \item 正の数列 \(\{a_n\}\) を
            \[
              a_1=1,\qquad
              a_{n+1}=\frac{a_n}{(1+2a_n)^2}
            \]
                によって定める。また
            \[
              b_n=\frac1{a_n}
            \]
                とする。
            \begin{enumerate}[label=(\arabic*)]
              \item \(n\ge2\) に対して \(b_n>4n-3\) を示せ。
              \item \(\displaystyle \lim_{n\to\infty}\frac1n\sum_{k=1}^{n}a_k\) を求めよ。
              \item \(\displaystyle \lim_{n\to\infty}na_n\) を求めよ。
            \end{enumerate}
                \smallskip
                \textit{参考構造：東京大学 2006 理科 第5問を参考に再構成}
          \item 正の整数 \(a_0\) から数列 \(\{a_n\}\) を
            \[
              a_{n+1}=
              \begin{cases}
                \dfrac{a_n}{2} & (a_n\text{ が偶数}),\\[4pt]
                \dfrac{3a_n+1}{2} & (a_n\text{ が奇数})
              \end{cases}
            \]
                によって定める。正整数 \(m\) に対し
            \[
              a_0,a_1,\ldots,a_m
            \]
                がすべて奇数となるような \(a_0\) の最小値を求めよ。
                \smallskip
                \textit{参考構造：京都大学 2024 理系 第4問を参考に再構成}
          \item 正整数 \(k\) に対して
            \[
              a_k=5^{\sqrt{k}}
            \]
                とする。\(a_k\) の整数部分が \(n\) 桁であるような \(k\) の個数を \(N_n\) とする。また,そのうち最高位の数字が \(2\) であるものの個数を \(L_n\) とする。
            \[
              \lim_{n\to\infty}\frac{L_n}{N_n}
            \]
                を求めよ。
                \smallskip
                \textit{参考構造：京都大学 2024 理系 第6問を参考に再構成}
          \item 数列 \(\{a_n\}\) を
            \[
              a_1=1,\qquad
              a_{n+1}=\frac{n}{n+1}a_n
            \]
                によって定める。正整数 \(m\) に対して
            \[
              S_m=\sum_{k=1}^{m}a_k a_{m-k+1}
            \]
                とする。\(S_m\) を求めよ。
                \smallskip
                \textit{参考構造：大阪大学 2025 文系 第2問の畳み込み構造を参考に再構成}
          \item 正方形の頂点を時計回りに \(A,B,C,D\) とする。点 \(P\) は初め \(A\) にあり,1秒ごとに独立に次の操作を行う。
            \[
              \begin{array}{c|c}
                \text{操作} & \text{確率}\\ \hline
                \text{その頂点にとどまる} & \frac12\\
                \text{時計回りに隣の頂点へ移る} & \frac14\\
                \text{反時計回りに隣の頂点へ移る} & \frac14
              \end{array}
            \]
                \(n\) 秒後に \(P\) が \(A\) にいる確率 \(p_n\) を求めよ。
                \smallskip
                \textit{参考構造：東京大学 2024 理系 第3問の状態分類・確率漸化式を参考に再構成}
        \end{enumerate}

      \subsubsection{方針}\label{節：大学過去問ベースのもの：方針}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-entrance}{hint}{problem}]
          \item 逆数列 \(b_n\) の差を計算する。まず \(a_n\) を上から評価して平均の極限を求め、その結果を再び \(b_n/n\) の極限評価に使う。
          \item 奇数項が続いている間は \(a_{n+1}+1=\frac32(a_n+1)\) が成立する。\(a_j\) が奇数である条件を \(a_0+1\) の2進因子の条件に読み替える。
          \item 桁数条件と最高位条件をそれぞれ \(k\) の区間へ変換する。整数個数と区間長との差は高々定数なので,極限では区間長の最高次項だけを比較する。
          \item まず階比型から \(a_n\) を求める。その後,畳み込みの各項を部分分数分解して調和和に変える。
          \item  \(A\),対頂点 \(C\),隣接頂点 \(B,D\) の3種類に状態分類する。\(A\) と \(C\) の確率の和と差を取る。
        \end{enumerate}

      \subsubsection{解答}\label{節：大学過去問ベースのもの：解答}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-entrance}{answer}{problem}]
          \item \[
              \lim_{n\to\infty}na_n=\frac14,
              \qquad
              \lim_{n\to\infty}\frac1n\sum_{k=1}^{n}a_k=0.
            \]
          \item \[
              a_0=2^{m+1}-1.
            \]
          \item \[
              \lim_{n\to\infty}\frac{L_n}{N_n}
              =\frac{d}{c}
              =\log_{10}\frac32
              .
            \]
          \item \[
              S_m=\frac2{m+1}\sum_{k=1}^{m}\frac1k
              .
            \]
          \item \[
              p_n=\frac14+\frac1{2^{n+1}}\quad(n\ge1)
              .
            \]
                なお \(p_0=1\) である.
        \end{enumerate}

      \subsubsection{解法}\label{節：大学過去問ベースのもの：解法}
        \begin{enumerate}[label=\textbf{問\arabic*.},leftmargin=3.8em,format=\bookitemlink{practice-entrance}{solution}{problem}]
          \item 漸化式から
            \[
              b_{n+1}=b_n+4+4a_n.
            \]
                したがって
            \[
              b_n=4n-3+4\sum_{k=1}^{n-1}a_k,
            \]
                ゆえに \(n\ge2\) で \(b_n>4n-3\). よって
            \[
              0<a_n<\frac1{4n-3},
            \]
                なので
            \[
              \frac1n\sum_{k=1}^{n}a_k\to0.
            \]
                さらに
            \[
              \frac{b_n}{n}
              =4-\frac3n+\frac4n\sum_{k=1}^{n-1}a_k\to4.
            \]
          \item 奇数項が続く間
            \[
              a_j+1=\frac{3^j}{2^j}(a_0+1).
            \]
                \(a_j\) が奇数であるためには \(a_j+1\) が偶数だから
            \[
              2^{j+1}\mid a_0+1.
            \]
                これが \(j=m\) まで成立する必要十分条件は
            \[
              2^{m+1}\mid a_0+1.
            \]
          \item  \(c=\log_5 10\) とすると
            \[
              N_n=c^2(2n-1)+O(1).
            \]
                最高位が2である条件は
            \[
              2\cdot10^{n-1}\le5^{\sqrt{k}}<3\cdot10^{n-1}.
            \]
                \(d=\log_5(\displaystyle \frac{3}{2})\) とおくと
            \[
              L_n=2cd\,n+O(1).
            \]
          \item 階比を掛け合わせると
            \[
              a_n=\frac1n.
            \]
                したがって
            \[
              S_m=\sum_{k=1}^{m}\frac1{k(m-k+1)}.
            \]
                ここで
            \[
              \frac1{k(m-k+1)}
              =\frac1{m+1}\left(\frac1k+\frac1{m-k+1}\right).
            \]
          \item  \(n\) 秒後に \(A,C\) にいる確率を \(p_n,q_n\),\(B,D\) のいずれかにいる確率を \(r_n\) とすると
            \[
              p_{n+1}=\frac12p_n+\frac14r_n,\qquad
              q_{n+1}=\frac12q_n+\frac14r_n.
            \]
                よって
            \[
              p_n-q_n=2^{-n}.
            \]
                また \(n\ge1\) で
            \[
              p_n+q_n=\frac12.
            \]
        \end{enumerate}
