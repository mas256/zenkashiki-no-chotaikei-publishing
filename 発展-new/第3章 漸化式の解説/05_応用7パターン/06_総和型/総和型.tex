\subsection{総和型}\label{節：総和型}
  \subsubsection{\texorpdfstring{総和と一般項の漸化式型 \, $S_n=pa_n+f(n)$}{総和と一般項の漸化式型}}
    ある漸化式によって定められる数列$a_n$によって,
    \[
      S_n=\sum_{k=1}^{n} a_k
    \]
    として定められる数列$S_n$について考えよう.
    $S_n$と$a_n$の関係を表した
    \[
      S_n=pa_n+f(n)
    \]
    のような漸化式を考える.
    このような漸化式は今まで取り扱った形にはない,未知のものである.
    であれば既知の形にするのが数学である.
    これまでのことをきちんと理解していれば,\,$a_{n+1}$と$a_n$の関係が分かっている時には解くことができる.
    $S_n$をうまく変形して$a_{n}$関連だけの式を産むには,\,$p \neq 1$の時,
    \begin{align*}
      &S_n=pa_n+f(n) \tag*{\(\cdots\) (1)}\\
      &S_{n+1}=pa_{n+1}+f(n+1) \tag*{\(\cdots\) (2)}
    \end{align*}
    $(2)-(1)$をして
    \begin{align*}
      &a_{n+1}=pa_{n+1}-pa_n+f(n+1)-f(n)\\
      &a_{n+1}-pa_{n+1}=-pa_n+f(n+1)-f(n)\\
      &a_{n+1}(1-p)=-pa_n+f(n+1)-f(n)\\
      &a_{n+1}=\frac{-pa_n+f(n+1)-f(n)}{(1-p)}\\
      &a_{n+1}=\frac{p}{p-1}a_n+\frac{1}{1-p}\{f(n+1)-f(n)\}
    \end{align*}
    $\displaystyle g(n)=\frac{1}{1-p}\{f(n+1)-f(n)\}$とおくと,
    \begin{align*}
      &a_{n+1}=\frac{p}{p-1}a_n+g(n)
    \end{align*}
    従って漸化式は階差等比複合型(\sectionref{節：階差等比複合型})に帰着する(係数$\frac{p}{p-1}$は$p=p-1$となることがないため,決して$1$にはならない).
    $p=1$のときはこの割り算ができない.この場合,$S_n=a_n+f(n)$と$S_n=S_{n-1}+a_n$($n\ge2$)を比較すると,
    \[
      a_{n-1}=f(n)-f(n-1)\quad(n\ge2)
    \]
    となる.したがって,
    \[
      a_n=f(n+1)-f(n)\quad(n\ge1)
    \]
    を得る.なお,$n=1$のもとの式から$f(1)=0$が必要である.

    \noindent \bookblocklink{example-summation-1}{problem}{answer}{【例題】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-1}{problem}{answer}]
      \item $\displaystyle S_n=2a_n-n$
      \item $\displaystyle S_n=\frac12a_n+n$
      \item $\displaystyle S_n=a_n+\binom n2$
    \end{enumerate}

    \noindent\bookblocklink{example-summation-1}{hint}{problem}{【方針】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-1}{hint}{problem}]
      \item まず$n=1$を代入して$a_1$を求めてから,本文の変形をなぞって$a_{n+1}$と$a_n$だけの漸化式を作りましょう.
      \item 同様です.できた漸化式は特性方程式型に帰着します.
      \item $p=1$の場合です.$S_n-S_{n-1}=a_n$と比較して,$a_{n-1}$を直接求めましょう.
    \end{enumerate}

    \noindent\bookblocklink{example-summation-1}{answer}{problem}{【解答】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-1}{answer}{problem}]
      \item $\displaystyle a_n=2^n-1$
      \item $\displaystyle a_n=1+(-1)^{n-1}$
      \item $\displaystyle a_n=n$
    \end{enumerate}

    \noindent\bookblocklink{example-summation-1}{solution}{problem}{【解法】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-1}{solution}{problem}]
    \item $\displaystyle S_n=2a_n-n$

        $n=1$を代入すると,
        \[
          S_1=a_1=2a_1-1
        \]
        より
        \[
          a_1=1
        \]
        また,
        \begin{align*}
          a_{n+1}
            &=S_{n+1}-S_n\\
            &=\{2a_{n+1}-(n+1)\}-(2a_n-n)\\
            &=2a_{n+1}-2a_n-1
        \end{align*}
        したがって,
        \[
          a_{n+1}=2a_n+1
        \]
        ここで $b_n=a_n+1$ とおくと,
        \begin{align*}
          b_{n+1}
            &=a_{n+1}+1\\
            &=2a_n+2\\
            &=2(a_n+1)\\
            &=2b_n
        \end{align*}
        また,
        \[
          b_1=a_1+1=2
        \]
        であるから,
        \[
          b_n=2^n
        \]
        よって,
        \[
          a_n=2^n-1
        \]
    \item $\displaystyle S_n=\frac12a_n+n$
        $n=1$を代入すると,
        \[
          S_1=a_1=\frac12a_1+1
        \]
        より
        \[
          a_1=2
        \]
        \begin{align*}
          a_{n+1}
            &=S_{n+1}-S_n\\
            &=\left\{\frac12a_{n+1}+(n+1)\right\}-\left(\frac12a_n+n\right)\\
            &=\frac12a_{n+1}-\frac12a_n+1
        \end{align*}
        したがって,
        \[
          a_{n+1}=-a_n+2
        \]

        ここで $b_n=a_n-1$ とおくと,
        \begin{align*}
          b_{n+1}
            &=a_{n+1}-1\\
            &=-a_n+1\\
            &=-(a_n-1)\\
            &=-b_n
        \end{align*}
        また,
        \[
          b_1=a_1-1=1
        \]
        であるから,
        \[
          b_n=(-1)^{n-1}
        \]
        よって,
        \[
          a_n=1+(-1)^{n-1}
        \]
    \item $\displaystyle S_n=a_n+\binom n2$

        $n=1$では
        \[
          S_1=a_1=a_1+\binom12
        \]
        となり,この式だけでは$a_1$は定まらない.

        $n\ge2$において,
        \begin{align*}
          a_n
            &=S_n-S_{n-1}\\
            &=\left(a_n+\binom n2\right)
              -\left(a_{n-1}+\binom{n-1}{2}\right)\\
            &=a_n-a_{n-1}+n-1
        \end{align*}
        であるから,
        \[
          a_{n-1}=n-1
        \]
        を得る.$n\ge2$で成り立つので,$n-1$を改めて$n$とおけば,
        \[
          a_n=n\quad(n\ge1)
        \]
        である.

    \end{enumerate}

  \subsubsection{\texorpdfstring{純総和型 \, $S_{n+1}=pS_n+f(n) $}{純総和型}}
    両辺に総和がきてもやることは一緒で,1つずらした総和と元の式を足し引きしてやることで,\,$a_n$や$a_{n+1}$の項を作れば良い.
    \begin{align*}
      &S_{n+1}=pS_n+f(n) \tag*{\(\cdots\) (1)}\\
      &S_{n}=pS_{n-1}+f(n-1) \tag*{\(\cdots\) (2)}\\
    \end{align*}
      $(1)-(2)$をして,
    \begin{align*}
      &a_{n+1}=pa_{n}+f(n)-f(n-1)
    \end{align*}
    ただし(2)は(1)の$n$を$n-1$に置き換えたものなので,この式は$n\ge2$の範囲でのみ成り立つ.\,$a_1$と$a_2$の関係は,別途$S_1,S_2$から直接求める必要がある.
    あとは階差等比複合型の計算である(\,$p=1$なら階差数列型になる).

    \noindent \bookblocklink{example-summation-2}{problem}{answer}{【例題】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-2}{problem}{answer}]
      \item $\displaystyle S_1=1,\ S_{n+1}=2S_n+n$
      \item $\displaystyle S_1=0,\ S_{n+1}=2S_n+n$
    \end{enumerate}

    \noindent\bookblocklink{example-summation-2}{hint}{problem}{【方針】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-2}{hint}{problem}]
      \item 本文の式は$n\ge2$でしか使えないことに注意し,$a_2$だけは$S_2=2S_1+1$から別に求めましょう.
      \item 同様に$n\ge2$で漸化式を作り,$a_2$から解きます.$a_1$も含む一般項になるか最後に確認しましょう.
    \end{enumerate}

    \noindent\bookblocklink{example-summation-2}{answer}{problem}{【解答】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-2}{answer}{problem}]
      \item $\displaystyle a_1=1,\quad a_n=3\cdot2^{n-2}-1\ (n\ge2)$
      \item $\displaystyle a_n=2^{n-1}-1\ (n\ge1)$
    \end{enumerate}

    \noindent\bookblocklink{example-summation-2}{solution}{problem}{【解法】}
    \begin{enumerate}[label=(\arabic*),parsep=3pt,format=\bookitemlink{example-summation-2}{solution}{problem}]
    \item $\displaystyle S_1=1,\ S_{n+1}=2S_n+n$

        $a_1=S_1=1$である.また,
        \[
          S_2=2S_1+1=3
        \]
        より
        \[
          a_2=S_2-S_1=2
        \]
        である.

        本文の公式
        \[
          a_{n+1}=pa_n+f(n)-f(n-1)\quad(n\ge2)
        \]
        に$p=2,\ f(n)=n$を代入すると,
        \[
          a_{n+1}=2a_n+1\quad(n\ge2)
        \]
        を得る.ここで$n\ge2$において$b_n=a_n+1$とおけば,
        \[
          b_{n+1}=2b_n,\qquad b_2=3
        \]
        であるから,
        \[
          b_n=3\cdot2^{n-2}\quad(n\ge2)
        \]
        したがって,
        \[
          a_1=1,\qquad a_n=3\cdot2^{n-2}-1\quad(n\ge2)
        \]
        である.

    \item $\displaystyle S_1=0,\ S_{n+1}=2S_n+n$

        $a_1=S_1=0$である.また,
        \[
          S_2=2S_1+1=1
        \]
        より
        \[
          a_2=S_2-S_1=1
        \]
        である.

        1と同様に,
        \[
          a_{n+1}=2a_n+1\quad(n\ge2)
        \]
        を得る.ここで$n\ge2$において$b_n=a_n+1$とおけば,
        \[
          b_{n+1}=2b_n,\qquad b_2=2
        \]
        であるから,
        \[
          b_n=2^{n-1}\quad(n\ge2)
        \]
        したがって,
        \[
          a_n=2^{n-1}-1\quad(n\ge2)
        \]
        となる.$n=1$のときも$a_1=0=2^0-1$であるから,
        \[
          a_n=2^{n-1}-1\quad(n\ge1)
        \]
        である.
    \end{enumerate}
